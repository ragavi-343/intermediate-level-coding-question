#include<stdio.h>
void duplicate(int arr[],int *n)
{
    int t[*n];
    int j=0;
    for(int i=0;i<*n;i++)
    {
        int dup=0;
        for(int k=0;k<j;k++)
        {
            if(arr[i]==t[k])
            {
                dup=1;
                break;
            }
        }
        if(!dup)
        {
            t[j++]=arr[i];
        }
    }
    for(int i=0;i<j;i++)
    {
        arr[i]=t[i];
    }
    *n=j;
}
int main(){
    int n;
    scanf("%d",&n);
    int arr[n];
    for(int i=0;i<n;i++)
    {
        scanf("%d",&arr[i]);
    }
    duplicate(arr, &n);
    for(int i=0;i<n;i++)
    {
        printf("%d ",arr[i]);
    }
    return 0;
}
